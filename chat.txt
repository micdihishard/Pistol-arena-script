-- Re-execution guard: an older copy must stop before it can fight the new copy
-- over the topbar chip, the GUI parent, or the chat feed.
local GENERATION_KEY = "__VEZ_CHAT_GENERATION"
local GENV = _G
pcall(function()
    if type(getgenv) == "function" then
        local g = getgenv()
        if type(g) == "table" then GENV = g end
    end
end)
local generation = 1
local generationShared = false
pcall(function()
    local prev = tonumber(GENV[GENERATION_KEY]) or 0
    generation = prev + 1
    GENV[GENERATION_KEY] = generation
    generationShared = true
end)
local function generationCurrent()
    if not generationShared then return true end
    local ok, cur = pcall(function() return tonumber(GENV[GENERATION_KEY]) end)
    return not ok or cur == nil or cur == generation
end

end

-- ================= hide Roblox's built-in chat window, keep its topbar slot =================
-- Disabling CoreGui chat removes the chat button entirely, and the other topbar
-- icons reflow into the gap. Keep the slot enabled and overlay our chip on it;
-- only the default chat window, input bar and channel tabs are disabled.
-- This does not stop TextChatService.MessageReceived or TextChannel:SendAsync.
local function hideDefaultChat()
    local slotOK = false
    pcall(function()
        local sg = game:GetService("StarterGui")
        local okGet, on = pcall(function()
            return sg:GetCoreGuiEnabled(Enum.CoreGuiType.Chat)
        end)
        if not okGet or on ~= true then
            sg:SetCoreGuiEnabled(Enum.CoreGuiType.Chat, true)
        end
        slotOK = true
    end)
    -- pcall success only means the closure ran; verify the slot actually reads enabled
    if slotOK then
        pcall(function()
            local sg = game:GetService("StarterGui")
            slotOK = sg:GetCoreGuiEnabled(Enum.CoreGuiType.Chat) == true
        end)
    end
    local cfgOK = false
    pcall(function()
        local cw = TCS:FindFirstChildOfClass("ChatWindowConfiguration")
        if cw and cw.Enabled then cw.Enabled = false end
        local ib = TCS:FindFirstChildOfClass("ChatInputBarConfiguration")
        if ib and ib.Enabled then ib.Enabled = false end
        local ct = TCS:FindFirstChildOfClass("ChannelTabsConfiguration")
        if ct and ct.Enabled then ct.Enabled = false end
        cfgOK = true
    end)
    return slotOK, cfgOK
end
local chatHiddenState
do
    local slotOK, cfgOK = hideDefaultChat()
    chatHiddenState = tostring(slotOK) .. ":" .. tostring(cfgOK)
    LOG("ui", "built-in chat window hidden, topbar slot kept: slot=%s configs=%s",
        tostring(slotOK), tostring(cfgOK))
end
-- Games can re-enable the default chat configs (or disable the CoreGui slot);
-- keep the slot alive and the window hidden without needing to re-execute.
task.spawn(function()
    local last = chatHiddenState
    while generationCurrent() do
        task.wait(2)
        if not generationCurrent() then break end
        local slotOK, cfgOK = hideDefaultChat()
        local state = tostring(slotOK) .. ":" .. tostring(cfgOK)
        if state ~= last then
            LOG("ui", "built-in chat config retry: slot=%s configs=%s", tostring(slotOK), tostring(cfgOK))
            last = state
        end
    end
end)

-- Last-resort recovery when the game/executor destroys our ScreenGui.  A
-- destroyed ScreenGui also destroys all of its descendants, so re-running the
-- source is the only way to rebuild the panel.  The generation guard above
-- makes the old copy exit instead of fighting the new one.
local RELOAD_KEY = "__VEZ_CHAT_LAST_RELOAD"
local lastReload = 0
local function reloadSelf(reason)
    if not generationCurrent() then return false end
    if type(loadstring) ~= "function" or type(readfile) ~= "function" then
        LOG("ui", "self reload unavailable (%s)", tostring(reason))
        return false
    end
    local now = os.time()
    local last = lastReload
    pcall(function() last = tonumber(GENV[RELOAD_KEY]) or last end)
    if now - last < 20 then return false end
    lastReload = now
    pcall(function() GENV[RELOAD_KEY] = now end)
    for _, path in ipairs(SCRIPT_PATHS) do
        local ok, src = pcall(readfile, path)
        if ok and type(src) == "string" and #src > 1000 then
            local fn, err = loadstring(src, "=VEZChatReload")
            if fn then
                LOG("ui", "self reload (%s) from %s", tostring(reason), path)
                local okRun, runErr = pcall(fn)
                if not okRun then LOG("ui", "self reload failed: %s", tostring(runErr)) end
                return true
            end
            LOG("ui", "self reload compile failed: %s", tostring(err))
        end
    end
    LOG("ui", "self reload source unavailable (%s)", tostring(reason))
    return false
end

-- ================= HTTP (executor `request` first, else HttpService) =================
local execReq = nil
pcall(function()
    if typeof(request) == "function" then execReq = request
    elseif typeof(http_request) == "function" then execReq = http_request end
end)
LOG("boot", "exec request fn: %s", execReq and "yes" or "no (HttpService only)")

local function doRequest(url, method, body)
    if execReq then
        local ok, res = pcall(execReq, { Url = url, Method = method, Headers = HEADERS, Body = body })
        if ok and res then
            local b = res.Body or res.body or ""
            local code = res.StatusCode or res.status_code or res.Status or 0
            local success = (code >= 200 and code < 300)
            -- Some executors omit StatusCode but still return a valid body.
            if code == 0 and b ~= "" and res.Success ~= false then success = true end
            if b ~= "" or success then
                return true, { Body = b, Success = success }
            else
                LOG("http", "execReq empty: code=%s url=%s", tostring(code), url)
            end
        else
            LOG("http", "execReq pcall fail: %s url=%s", tostring(res), url)
        end
    end
    local ok2, res2 = pcall(function()
        return HttpService:RequestAsync({ Url = url, Method = method, Headers = HEADERS, Body = body })
    end)
    if ok2 and res2 then return true, res2 end
    LOG("http", "RequestAsync fail: %s url=%s", tostring(res2), url)
    return false, res2
end

-- Old failure: PlayerGui not replicated yet -> parent nil -> invisible GUI.
local function resolveParent()
    local pg = nil
    pcall(function()
        if typeof(gethui) == "function" then pg = gethui()
        elseif typeof(get_hui) == "function" then pg = get_hui() end
    end)
    if typeof(pg) == "Instance" then return pg, "gethui" end
    pcall(function() pg = lp:FindFirstChildOfClass("PlayerGui") end)
    if typeof(pg) == "Instance" then return pg, "PlayerGui" end
    return nil, "none"
end

    pcall(function() lp:WaitForChild("PlayerGui", 30) end)

    parentGui, parentHow = resolveParent()
end

    pcall(function() push(lp:FindFirstChildOfClass("PlayerGui")) end)
    pcall(function() if typeof(gethui) == "function" then push(gethui()) end end)
    pcall(function() if typeof(get_hui) == "function" then push(get_hui()) end end)
    pcall(function() push(game:GetService("CoreGui")) end)
    for _, r in ipairs(roots) do
        local okChildren, children = pcall(function() return r:GetChildren() end)
        if okChildren and children then
            for _, c in ipairs(children) do
                if c.Name == "VEZChat" then
                    pcall(function() c:Destroy() end)
                    local where = "?"
                    pcall(function() where = r:GetFullName() end)
                    LOG("ui", "removed duplicate VEZChat from %s", where)
                end
            end
        end
        local okDesc, list = pcall(function() return r:GetDescendants() end)
        if okDesc and list then
            for _, d in ipairs(list) do
                if d.Name == "VEZChatTopbar" then
                    pcall(function() d:Destroy() end)
                    local where = "?"
                    pcall(function() where = r:GetFullName() end)

-- if PlayerGui arrives late / executor hui swaps, re-parent so we never stay invisible
task.spawn(function()
    local warned = false
    local failures = 0
    local enabledWarned = false
    while generationCurrent() do
        task.wait(2)
        if not generationCurrent() then break end
        local ok, parent = pcall(function() return gui.Parent end)
        if not ok or not parent then
            failures = failures + 1
            local np, how = resolveParent()
            if typeof(np) == "Instance" then
                local okp = pcall(function() gui.Parent = np end)
                if okp then
                    LOG("ui", "re-parented GUI to %s (%s)", np:GetFullName(), how)
                    warned, failures, enabledWarned = false, 0, false
                elseif not warned then
                    warned = true
                    LOG("ui", "GUI has no parent and cannot be re-parented (%s)", how)
                end
            elseif not warned then
                warned = true
                LOG("ui", "GUI has no parent and no UI parent is available")
            end
            if failures >= 3 then
                failures = 0
                task.spawn(function() reloadSelf("ScreenGui destroyed or unparentable") end)
            end
        else
            warned, failures = false, 0
            pcall(function()
                if gui.Enabled == false then
                    gui.Enabled = true
                    if not enabledWarned then
                        enabledWarned = true
                        LOG("ui", "GUI was disabled; re-enabled it")
                    end
                else
                    enabledWarned = false
                end
            end)
        end
    end
end)

local chatIconImage = nil
do
    local function b64decode(s)
        local chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
        local map = {}
        for i = 1, #chars do map[chars:sub(i, i)] = i - 1 end
        s = s:gsub("%s+", "")
        local out = {}
        for i = 1, #s, 4 do
            local c1, c2 = s:sub(i, i), s:sub(i + 1, i + 1)
            local c3, c4 = s:sub(i + 2, i + 2), s:sub(i + 3, i + 3)
            local v1, v2 = map[c1] or 0, map[c2] or 0
            local v3 = c3 ~= "=" and (map[c3] or 0) or nil
            local v4 = c4 ~= "=" and (map[c4] or 0) or nil
            local v = v1 * 262144 + v2 * 4096 + (v3 or 0) * 64 + (v4 or 0)
            out[#out + 1] = string.char(math.floor(v / 65536) % 256)
            if v3 then out[#out + 1] = string.char(math.floor(v / 256) % 256) end
            if v4 then out[#out + 1] = string.char(v % 256) end
        end
        return table.concat(out)
    end
    local ok, res = pcall(function()
        local bytes = b64decode(ICON_B64)
        local name = "vez_chat_icon.png"
        local old = nil
        if type(isfile) == "function" and isfile(name) and type(readfile) == "function" then
            old = readfile(name)
        end
        if type(writefile) == "function" and old ~= bytes then writefile(name, bytes) end
        if type(getcustomasset) == "function" then return getcustomasset(name) end
        if type(getsynasset) == "function" then return getsynasset(name) end
        error("getcustomasset/getsynasset unavailable")
    end)
    local valid = ok and type(res) == "string"
        and (res:match("^rbxasset") ~= nil or res:match("^https?://") ~= nil or res:match("^file://") ~= nil)
    if valid then
        chatIconImage = res
        LOG("ui", "chat icon asset ready (%d b64 chars)", #ICON_B64)
    else
        LOG("ui", "chat icon failed (%s); keeping CHAT text", tostring(res))
    end
end

-- ================= topbar slot overlay =================
-- The built-in chat button is kept (see hideDefaultChat) purely as the layout
-- host for our chip. We hide its artwork rather than the button itself so the
-- topbar keeps reserving the same space.
local topbarChip, topbarHost = nil, nil

local function getCoreGui()
    local cg = nil
    pcall(function() cg = game:GetService("CoreGui") end)
    if not cg then pcall(function() cg = game.CoreGui end) end
    return cg
end

local function isUnder(o, root)
    local node = o
    for _ = 1, 12 do
        if node == nil then return false end
        if node == root then return true end
        local ok, p = pcall(function() return node.Parent end)
        if not ok then return false end
        node = p
    end
    return false
end

local function objectImage(o)
    local ok, img = pcall(function() return o.Image end)
    if ok and type(img) == "string" then return img end
    return ""
end

-- Never treat a hidden "Chat" entry in the avatar/settings dropdown as the
-- topbar chat slot.  Walking up from such an entry used to land on the
-- player's headshot button, which put our 42px chip over the avatar.
local TOPBAR_BAD = {
    "headshot", "avatar", "profile", "playerlist", "leaderboard",
    "friends", "party", "platform", "settings", "emote",
    "backpack", "inventory", "shop", "robux", "notification",
    "hamburger", "home", "voice", "microphone", "leave", "report",
    "dropdown", "contextmenu", "popup",
}
local function topbarHostBad(full)
    full = tostring(full or ""):lower()
    for _, token in ipairs(TOPBAR_BAD) do
        if full:find(token, 1, true) then return token end
    end
    return nil
end

local function scoreChatCandidate(o)
    local score = 0
    local name = ""
    pcall(function() name = o.Name end)
    name = tostring(name):lower()
    if name == "vezchattopbar" then return -100 end
    if name:find("chat", 1, true) then
        if name == "chat" or name == "chaticon" or name == "chatbutton" then
            score = score + 60
        else
            score = score + 30
        end
    end
    local img = objectImage(o):lower()
    if img:find("chat", 1, true) then score = score + 25 end
    local full = ""
    pcall(function() full = o:GetFullName():lower() end)
    if topbarHostBad(full) then score = score - 100 end
    if full:find("topbarapp", 1, true) then score = score + 3 end
    if full:find("leftframe", 1, true) then score = score + 5 end
    if full:find("friends", 1, true) or full:find("party", 1, true)
        or full:find("platform", 1, true) then
        score = score - 60
    end
    -- never mistake the (hidden) default chat window/input for the topbar button
    if full:find("chatwindow", 1, true) or full:find("chatinput", 1, true)
        or full:find("chatframe", 1, true) or full:find("chatbar", 1, true) then
        score = score - 80
    end
    return score
end

local function buttonFor(o, stopAt)
    local node = o
    for _ = 1, 6 do
        if node == nil or node == stopAt then break end
        local ok, isBtn = pcall(function() return node:IsA("GuiButton") end)
        if ok and isBtn then return node end
        local okp, p = pcall(function() return node.Parent end)
        if not okp then break end
        node = p
    end
    return o
end

local function visibleChain(o)
    local node = o
    for _ = 1, 12 do
        if node == nil then break end
        local ok, vis = pcall(function() return node.Visible end)
        if ok and vis == false then return false end
        local oken, en = pcall(function() return node.Enabled end)
        if oken and en == false then return false end
        local okp, p = pcall(function() return node.Parent end)
        if not okp then break end
        node = p
    end
    return true
end

local function hostHasChatArt(host)
    local name = ""
    pcall(function() name = tostring(host.Name):lower() end)
    if name:find("chat", 1, true) then return true end
    if objectImage(host):lower():find("chat", 1, true) then return true end
    local ok, list = pcall(function() return host:GetDescendants() end)
    if ok and list then
        for _, d in ipairs(list) do
            local dn = ""
            pcall(function() dn = tostring(d.Name):lower() end)
            if (dn:find("chat", 1, true)
                or objectImage(d):lower():find("chat", 1, true))
                and visibleChain(d) then
                return true
            end
        end
    end
    return false
end

local function topbarHostValid(host)
    if not host then return false end
    local okp, parent = pcall(function() return host.Parent end)
    if not okp or not parent then return false end
    local okGui, isGui = pcall(function() return host:IsA("GuiObject") end)
    if not okGui or not isGui then return false end
    local full, name = "", ""
    pcall(function() full = host:GetFullName() end)
    pcall(function() name = host.Name end)
    full = tostring(full):lower()
    name = tostring(name):lower()
    if full:find("vezchattopbar", 1, true) then return false end
    if topbarHostBad(full) then return false end
    -- The legacy ChatIcon path is authoritative.  Accept it even if a client
    -- renamed an ancestor or reports an unexpected AbsoluteSize.
    if full:find("chaticon", 1, true) then return true end
    if not (full:find("topbar", 1, true) or full:find("unibar", 1, true)
        or full:find("chrome", 1, true) or full:find("leftframe", 1, true)
        or full:find("experience", 1, true) or full:find("hud", 1, true)) then
        return false
    end
    if not hostHasChatArt(host) then return false end
    local sx, sy = 0, 0
    pcall(function()
        local s = host.AbsoluteSize
        if s then sx, sy = s.X, s.Y end
    end)
    -- A topbar icon is small.  A large container with a chat descendant is
    -- not the slot; attaching there centers the chip over other topbar UI.
    if (sx > 96 or sy > 96) and name ~= "chat" and name ~= "chaticon"
        and name ~= "chatbutton" then
        return false
    end
    return true
end

local function scanChatHost(root, threshold, requireTopbarAncestry)
    local best, bestScore = nil, threshold
    local ok, list = pcall(function() return root:GetDescendants() end)
    if not ok or not list then return nil end
    for _, o in ipairs(list) do
        local okGui, isGui = pcall(function() return o:IsA("GuiObject") end)
        if okGui and isGui then
            local full = ""
            pcall(function() full = o:GetFullName():lower() end)
            local topbarish = full:find("topbar", 1, true) or full:find("unibar", 1, true)
                or full:find("chrome", 1, true) or full:find("experience", 1, true)
                or full:find("leftframe", 1, true) or full:find("hud", 1, true)
            if topbarish or not requireTopbarAncestry then
                local s = scoreChatCandidate(o)
                -- Hidden dropdown entries are not the topbar slot.
                if s > 20 and visibleChain(o) then
                    s = s + 40
                    if s > bestScore then best, bestScore = o, s end
                end
            end
        end
    end
    return best
end

local lastTopbarReject = nil
local function noteTopbarReject(host)
    local full = "?"
    pcall(function() full = host:GetFullName() end)
    full = tostring(full)
    if full == lastTopbarReject then return end
    lastTopbarReject = full
    LOG("ui", "topbar host rejected: %s", full)
end

local function findTopbarChatHost()
    local cg = getCoreGui()
    if not cg then return nil end
    local top = nil
    pcall(function() top = cg:FindFirstChild("TopBarApp") end)
    if top then
        local legacy = nil
        pcall(function()
            local tf = top:FindFirstChild("TopBarFrame")
            local lf = tf and tf:FindFirstChild("LeftFrame")
            legacy = lf and lf:FindFirstChild("ChatIcon")
        end)
        if legacy and legacy:IsA("GuiObject") then
            local cand = buttonFor(legacy, cg)
            if visibleChain(cand) then
                if topbarHostValid(cand) then return cand end
                noteTopbarReject(cand)
            end
        end
    end
    local best = top and scanChatHost(top, 20, false) or nil
    if not best then
        -- Some clients do not keep chat inside TopBarApp. Scan CoreGui too,
        -- but only under topbar/unibar/chrome-like ancestors so the hidden
        -- default chat window cannot be mistaken for the icon.
        best = scanChatHost(cg, 20, true)
    end
    if not best then return nil end
    local host = buttonFor(best, cg)
    if visibleChain(host) and topbarHostValid(host) then return host end
    if visibleChain(best) and topbarHostValid(best) then return best end
    noteTopbarReject(host or best)
    return nil
end

local function hideTopbarArt(host, chip)
    local function hideOne(o)
        if o == nil or o == chip then return end
        if isUnder(o, chip) then return end
        -- never hide another VEZ chip; a stale copy may still be present
        if o.Name == "VEZChatTopbar" then return end
        pcall(function()
            if o:IsA("ImageLabel") or o:IsA("ImageButton") then o.ImageTransparency = 1 end
            if o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox") then o.TextTransparency = 1 end
            if o:IsA("Frame") or o:IsA("ScrollingFrame") or o:IsA("ViewportFrame") then
                o.BackgroundTransparency = 1
            end
            if o:IsA("UIStroke") then o.Transparency = 1 end
            -- our chip is the only click target; stop the built-in button and
            -- any invisible touch frame from intercepting taps on the slot
            if o:IsA("GuiButton") then
                pcall(function() o.Interactable = false end)
            end
            if o:IsA("GuiObject") then
                pcall(function() o.Active = false end)
                pcall(function() o.Selectable = false end)
            end
        end)
    end
    hideOne(host)
    local ok, list = pcall(function() return host:GetDescendants() end)
    if ok and list then
        for _, d in ipairs(list) do hideOne(d) end
    end
end

local function syncTopbarChipSize()
    if not (topbarChip and topbarHost) then return end
    local size = FLOAT_SIZE
    local ok, hs = pcall(function() return topbarHost.AbsoluteSize end)
    if ok and hs and hs.X and hs.Y then
        local m = math.min(FLOAT_SIZE, hs.X, hs.Y)
        if m >= 24 then size = math.floor(m + 0.5) end
    end
    local current = -1
    pcall(function() current = topbarChip.Size.X.Offset end)
    if current == size then return end
    topbarChip.Size = UDim2.fromOffset(size, size)
    local corner = topbarChip:FindFirstChildOfClass("UICorner")
    if corner then corner.CornerRadius = UDim.new(0, math.floor(size * 0.43)) end
end

-- A stale copy of this script can hide/re-disable our chip every couple of
-- seconds.  Re-assert our own visual/input state so the current copy stays usable.
local function restoreTopbarChip()
    local chip = topbarChip
    if not chip then return end
    pcall(function()
        chip.Visible = true
        chip.Active = true
        chip.Selectable = true
        chip.BackgroundTransparency = 0
        chip.TextTransparency = 0
        if chip:IsA("GuiButton") then chip.Interactable = true end
        local icon = chip:FindFirstChild("Icon")
        if icon then icon.ImageTransparency = 0 end
        local stroke = chip:FindFirstChildOfClass("UIStroke")
        if stroke then stroke.Transparency = 0.75 end
        local badge = chip:FindFirstChild("Unread")
        if badge then badge.TextTransparency = 0 end
    end)
end

local function topbarAttached()
    if not (topbarChip and topbarHost) then return false end
    local ok1, cp = pcall(function() return topbarChip.Parent end)
    if not ok1 or cp ~= topbarHost then return false end
    local ok2, hp = pcall(function() return topbarHost.Parent end)
    if not ok2 or hp == nil then return false end
    return topbarHostValid(topbarHost)
end

local function detachTopbarChip()
    if topbarBadge then
        pcall(function() topbarBadge:Destroy() end)
        topbarBadge = nil
    end
    if topbarChip then pcall(function() topbarChip:Destroy() end) end
    topbarChip, topbarHost = nil, nil
    floatBtn.Visible = true
    refreshFloat()
end

local function attachTopbarChip()
    if topbarChip and topbarHost then
        if topbarAttached() then
            syncTopbarChipSize()
            return true
        end
        detachTopbarChip()
    end
    local host = findTopbarChatHost()
    if not host or not topbarHostValid(host) then return false end
    local ok, err = pcall(function()
        local chip = buildTopbarChip()
        topbarChip, topbarHost = chip, host
        chip.Parent = host
        hideTopbarArt(host, chip)
        syncTopbarChipSize()
    end)
    if not ok then
        LOG("ui", "topbar chip attach failed: %s", tostring(err))
        detachTopbarChip()
        return false
    end
    task.defer(function() syncTopbarChipSize() end)
    floatBtn.Visible = not visibleChain(topbarHost)
    local full = "?"
    pcall(function() full = topbarHost:GetFullName() end)
    LOG("ui", "topbar chip attached to %s (pill visible=%s)",
        tostring(full), tostring(floatBtn.Visible))
    refreshFloat()
    return true
end

do
    if attachTopbarChip() then
        LOG("ui", "floating pill hidden (topbar slot active)")
    else
        LOG("ui", "topbar chat slot not found yet; floating pill active")
    end
end
task.spawn(function()
    local misses = 0
    while generationCurrent() do
        task.wait(2)
        if not generationCurrent() then break end
        if topbarAttached() then
            misses = 0
            syncTopbarChipSize()
            -- remove stale copies of the chip left by an older execution
            local okList, list = pcall(function() return topbarHost:GetDescendants() end)
            if okList and list then
                for _, d in ipairs(list) do
                    if d.Name == "VEZChatTopbar" and d ~= topbarChip then
                        pcall(function() d:Destroy() end)
                    end
                end
            end
            -- built-in chat may repaint its icon/badge; keep its art hidden
            hideTopbarArt(topbarHost, topbarChip)
            restoreTopbarChip()
            pcall(function() if topbarHost.ZIndex ~= 1000 then topbarHost.ZIndex = 1000 end end)
            pcall(function() if topbarChip.ZIndex ~= 1000 then topbarChip.ZIndex = 1000 end end)
            -- if the topbar itself is hidden, keep the floating pill as fallback
            local showPill = not visibleChain(topbarHost)
            pcall(function() if floatBtn.Visible ~= showPill then floatBtn.Visible = showPill end end)
        else
            if topbarChip or topbarHost then
                LOG("ui", "topbar chip lost; falling back to floating pill")
                detachTopbarChip()
            end
            pcall(function() floatBtn.Visible = true end)
            misses = misses + 1
            if misses <= 30 then
                if attachTopbarChip() then
                    misses = 0
                    LOG("ui", "floating pill hidden (topbar slot active)")
                elseif misses == 30 then
                    LOG("ui", "topbar chat slot unavailable after retries; keeping floating pill, retrying slowly")
                end
            elseif misses % 5 == 0 then
                if attachTopbarChip() then
                    misses = 0
                    LOG("ui", "floating pill hidden (topbar slot active)")
                end
            end
        end
    end
end)

local function topbarTapArea(input)
    if not generationCurrent() then return end
    local t = input.UserInputType
    if t ~= Enum.UserInputType.Touch and t ~= Enum.UserInputType.MouseButton1 then return end
    local chip = topbarChip
    if not chip then return end
    if not topbarAttached() then return end
    -- Do not require the full visible chain here: this is the fallback for a
    -- chip that the game/old copy made transparent or non-interactable.
    local okVis, vis = pcall(function()
        return chip.Visible ~= false and (not topbarHost or topbarHost.Visible ~= false)
    end)
    if not okVis or not vis then return end
    local ap, sz
    pcall(function()
        ap = chip.AbsolutePosition
        sz = chip.AbsoluteSize
    end)
    local pos = input.Position
    if not ap or not sz or not pos then return end
    local insetY = 0
    pcall(function() insetY = game:GetService("GuiService"):GetGuiInset().Y end)
    local px, py = pos.X, pos.Y
    local function inside(x, y)
        return x >= ap.X - 6 and x <= ap.X + sz.X + 6
            and y >= ap.Y - 6 and y <= ap.Y + sz.Y + 6
    end
    if inside(px, py) or inside(px, py - insetY) then
        LOG("ui", "topbar area tapped (input fallback)")
        chipTap("topbar-area")
    end
end
UIS.InputBegan:Connect(topbarTapArea)

-- viewport helper: read camera FRESH every call (old code cached it once -> bad clamp after rotation/resize)
local function viewport()
    local cam = nil
    pcall(function() cam = workspace.CurrentCamera end)
    if cam then return cam.ViewportSize end
    return Vector2.new(800, 600)
end

local function panelAbs()
    local vs = viewport()
    local p = main.Position
    return p.X.Scale * vs.X + p.X.Offset, p.Y.Scale * vs.Y + p.Y.Offset
end
local function clampPanelPos(ax, ay)
    local vs = viewport()
    local w, h = main.AbsoluteSize.X, main.AbsoluteSize.Y
    if w <= 0 then w = MIN_W end
    if h <= 0 then h = MIN_H end
    return UDim2.new(0, math.clamp(ax, 0, math.max(0, vs.X - w)),
        0, math.clamp(ay, 0, math.max(0, vs.Y - h)))
end
do
    local ax, ay = panelAbs()
    main.Position = clampPanelPos(ax, ay)
end
local function panelOnScreen()
    local ax, ay = panelAbs()
    local vs = viewport()
    local w, h = main.AbsoluteSize.X, main.AbsoluteSize.Y
    if w <= 0 then w = MIN_W end
    if h <= 0 then h = MIN_H end
    return ax + w > 0 and ay + h > 0 and ax < vs.X and ay < vs.Y
end

local panelParentFailures = 0
local function ensurePanelParented()
    local okp, parent = pcall(function() return gui.Parent end)
    if okp and parent then
        panelParentFailures = 0
        return true, false
    end
    local np, how = resolveParent()
    if typeof(np) == "Instance" then
        local ok = pcall(function() gui.Parent = np end)
        if ok then
            LOG("ui", "panel GUI re-parented to %s (%s)", np:GetFullName(), how)
            panelParentFailures = 0
            return true, true
        end
    end
    panelParentFailures = panelParentFailures + 1
    LOG("ui", "panel GUI is unavailable and could not be re-parented")
    if panelParentFailures >= 3 then
        panelParentFailures = 0
        task.spawn(function() reloadSelf("panel GUI unavailable") end)
    end
    return false, false
end

togglePanel = function(source)
    if not generationCurrent() then return end
    local parented, reparented = ensurePanelParented()
    if not parented then return end
    local wasEnabled = true
    pcall(function() wasEnabled = gui.Enabled ~= false end)
    pcall(function() gui.Enabled = true end)
    pcall(function() gui.DisplayOrder = 999999 end)
    -- The frame can be unparented independently of the ScreenGui.
    local mainReparented = false
    local okMain, mainErr = pcall(function()
        if main.Parent ~= gui then
            main.Parent = gui
            mainReparented = true
        end
    end)
    if not okMain then
        LOG("ui", "panel frame unavailable: %s", tostring(mainErr))
        task.spawn(function() reloadSelf("panel frame unavailable") end)
        return
    end
    local wasVisible, onScreen = false, false
    pcall(function() wasVisible = main.Visible == true end)
    pcall(function() onScreen = panelOnScreen() end)
    -- If the GUI was disabled, the panel was not actually visible even though
    -- main.Visible was still true.  The first tap must open it, not hide it.
    if wasVisible and wasEnabled and not reparented and not mainReparented and onScreen then
        pcall(function() main.Visible = false end)
        LOG("ui", "%s tapped -> panel hidden", tostring(source))
        return
    end
    pcall(function() main.Position = clampPanelPos(panelAbs()) end)
    local okShow = pcall(function() main.Visible = true end)
    if not okShow then
        LOG("ui", "panel frame could not be shown")
        task.spawn(function() reloadSelf("panel frame could not be shown") end)
        return
    end
    unread = 0
    pcall(refreshFloat)
    if renderFeed then
        local ok, err = pcall(renderFeed, activeTab)
        if not ok then LOG("ui", "renderFeed failed on open: %s", tostring(err)) end
    end
    scrollMsgsBottom()
    LOG("ui", "%s tapped -> panel shown", tostring(source))
end

-- ================= persist locked UI positions across re-executes =================
-- /lock stores pill + panel positions and the lock flag (tiny comment markers
-- appended to this script + a sidecar file); on boot they are restored and clamped.
local function parsePillPos(s)
    if type(s) ~= "string" then return nil end
    local sx, sy, ox, oy = s:match("^([%-%d%.]+)|([%-%d%.]+)|([%-%d%.]+)|([%-%d%.]+)$")
    if not sx then return nil end
    local n1, n2, n3, n4 = tonumber(sx), tonumber(sy), tonumber(ox), tonumber(oy)
    if not (n1 and n2 and n3 and n4) then return nil end
    if n1 < 0 or n1 > 1 or n2 < 0 or n2 > 1 then return nil end
    return UDim2.new(n1, n3, n2, n4)
end

-- saved payload: pill4 [| panel4]; older saves contain only the pill4 part
local function parseSavedPos(s)
    if type(s) ~= "string" then return nil end
    local p = {}
    for piece in s:gmatch("[^|]+") do p[#p + 1] = piece end
    if #p == 4 then
        local pill = parsePillPos(s)
        if pill then return { pill = pill } end
    elseif #p == 8 then
        local pill = parsePillPos(table.concat(p, "|", 1, 4))
        local panel = parsePillPos(table.concat(p, "|", 5, 8))
        if pill and panel then return { pill = pill, panel = panel } end
    end
    return nil
end

local function lastSavedPill(text)
    if type(text) ~= "string" then return nil, 0 end
    local lastVal, lastTs = nil, 0
    for v, ts in text:gmatch("%-%- PILLPOS ([%d%.%-|]+) (%d+)") do
        lastVal, lastTs = v, tonumber(ts) or 0
    end
    return lastVal, lastTs
end

local function lastSavedLock(text)
    if type(text) ~= "string" then return nil, -1 end
    local lock, lastTs = nil, -1
    for v, ts in text:gmatch("%-%- LOCKED ([01]) (%d+)") do
        lock, lastTs = (v == "1"), tonumber(ts) or 0
    end
    return lock, lastTs
end

local function readSavedPill()
    local bestPos, bestTs, bestFrom = nil, -1, nil
    local bestLock, bestLockTs = nil, -1
    local function consider(path, label)
        local ok, text = pcall(readfile, path)
        if not ok then return end
        local val, ts = lastSavedPill(text)
        local pos = parseSavedPos(val)
        if pos and ts >= bestTs then
            bestPos, bestTs, bestFrom = pos, ts, label .. ":" .. path
        end
        local lock, lts = lastSavedLock(text)
        if lock ~= nil and lts >= bestLockTs then
            bestLock, bestLockTs = lock, lts
        end
    end
    for _, path in ipairs(SCRIPT_PATHS) do consider(path, "script") end
    for _, path in ipairs(SIDECAR_PATHS) do consider(path, "sidecar") end
    return bestPos, bestFrom, bestLock
end

local function posStr(u)
    return string.format("%.4f|%.4f|%.0f|%.0f", u.X.Scale, u.Y.Scale, u.X.Offset, u.Y.Offset)
end

local function currentPillStr()
    return posStr(floatBtn.Position) .. "|" .. posStr(main.Position)
end

local function savePillPos()
    local val = currentPillStr()
    local ts = tostring(os.time())
    local data = "-- PILLPOS " .. val .. " " .. ts
        .. "\n-- LOCKED " .. (locked and "1" or "0") .. " " .. ts
    local okScript, okSide = false, false
    for _, path in ipairs(SIDECAR_PATHS) do
        if pcall(writefile, path, data) then okSide = true end
    end
    if type(appendfile) == "function" then
        for _, path in ipairs(SCRIPT_PATHS) do
            local ok = pcall(function()
                local before = readfile(path)
                appendfile(path, "\n" .. data .. "\n")
                local after = readfile(path)
                local kept = type(after) == "string" and #after >= #before + #data
                    and after:sub(1, #before) == before
                if not kept then pcall(writefile, path, before) end
                return kept and lastSavedPill(after) == val
            end)
            if ok then okScript = true end
        end
    end
    LOG("persist", "save ui pos=%s locked=%s script=%s sidecar=%s",
        val, tostring(locked), tostring(okScript), tostring(okSide))
    if okScript and okSide then return "script+sidecar" end
    if okScript then return "script" end
    if okSide then return "sidecar" end
    return "none"
end

do
    local okRead, saved, from, savedLock = pcall(readSavedPill)
    if okRead and saved then
        if saved.panel then main.Position = saved.panel end
        if saved.pill then floatBtn.Position = saved.pill end
        -- older saves (no LOCKED marker) were only written by /lock
        if savedLock == nil then savedLock = true end
        locked = savedLock
        LOG("persist", "restored ui pos from %s (locked=%s)",
            tostring(from), tostring(locked))
    end
    local ax, ay = floatAbs()
    floatBtn.Position = clampFloatPos(ax, ay)
    local px, py = panelAbs()
    main.Position = clampPanelPos(px, py)
end

    floatBtn.InputBegan:Connect(function(input)
        if not generationCurrent() then return end
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            fDrag, fStart, fMoved = true, input.Position, 0
            fBtnAx, fBtnAy = floatAbs()
            LOG("drag", "float drag start (locked=%s)", tostring(locked))
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if not generationCurrent() then return end
        if not fDrag then return end
        local st = input.UserInputState
        if st == Enum.UserInputState.End or st == Enum.UserInputState.Cancel then return end
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseMovement or t == Enum.UserInputType.Touch then
            local d = input.Position - fStart
            fMoved = d.Magnitude
            if fMoved > 4 and not locked then
                pcall(function() floatBtn.Position = clampFloatPos(fBtnAx + d.X, fBtnAy + d.Y) end)
            end
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if not generationCurrent() then return end
        if not fDrag then return end
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            fDrag = false
            if locked then
                -- position frozen: snap back, tap still toggles panel
                pcall(function() floatBtn.Position = clampFloatPos(fBtnAx, fBtnAy) end)
                if fMoved <= 10 then
                    if togglePanel then
                        togglePanel("float locked")
                    else
                        main.Visible = not main.Visible
                        LOG("ui", "float tapped (locked) -> panel visible=%s", tostring(main.Visible))
                    end
                else
                    LOG("drag", "float move blocked (locked)")
                end
            elseif fMoved <= 10 then
                if togglePanel then
                    togglePanel("float")
                else
                    main.Visible = not main.Visible
                    LOG("ui", "float tapped -> panel visible=%s", tostring(main.Visible))
                end
            else
                local ax, ay = floatAbs()
                LOG("drag", "float dropped at %d,%d", math.floor(ax), math.floor(ay))
            end
        end
    end)
end

-- Real Roblox chat name colors: deterministic from the USERNAME (seed) using
-- the TextChatService palette (Bright violet was replaced by Alder in 2022).
-- Team members use their team color instead, matching Roblox's core chat.
local NAME_COLORS = {
    Color3.fromRGB(253, 41, 67),
    Color3.fromRGB(1, 162, 255),
    Color3.fromRGB(2, 184, 87),
    Color3.fromRGB(180, 128, 255),
    Color3.fromRGB(218, 133, 65),
    Color3.fromRGB(245, 205, 48),
    Color3.fromRGB(232, 186, 200),
    Color3.fromRGB(215, 197, 154),
}
local function rawNameColor(seed)
    local value = 0
    local n = #seed
    for i = 1, n do
        local c = string.byte(seed, i)
        local rev = n - i + 1
        if n % 2 == 1 then rev = rev - 1 end
        if rev % 4 >= 2 then c = -c end
        value = value + c
    end
    return NAME_COLORS[(value % #NAME_COLORS) + 1]
end
local function nameColor(seed, plr)
    if plr then
        local ok, c = pcall(function()
            if plr.Team ~= nil then return plr.TeamColor.Color end
        end)
        if ok and c then return c end
    end
    return rawNameColor(seed or "Guest")
end
local function esc(s)
    return s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")
end

local function renderBubble(name, text, isMe, color, isSystem)
    order = order + 1

        pad.PaddingBottom = UDim.new(0, isSystem and 1 or 2)
        pad.PaddingLeft = UDim.new(0, isSystem and 2 or 3)
        pad.PaddingRight = UDim.new(0, 3)
        pad.Parent = f
        local c = color or nameColor(isMe and lp.Name or name, isMe and lp or nil)

-- store into a feed; render only when the active panel is visible, otherwise
-- bump unread. Rendering while hidden made long history loads lag badly.
-- seed = username used for the real name color; plr enables the team-color override
-- opts: { system = true, color = Color3, silent = true (no unread badge) }
local function panelVisible()
    local ok, visible = pcall(function()
        return main.Visible == true and gui.Enabled ~= false
    end)
    return ok and visible
end

local function storeMsg(feed, name, text, isMe, seed, plr, opts)
    if type(text) ~= "string" or text:match("^%s*$") then return end
    if type(name) ~= "string" or name == "" then name = "Guest" end
    local f = feeds[feed]
    if not f then LOG("msg", "bad feed %s", tostring(feed)) return end
    opts = opts or {}
    local color = opts.color or nameColor(seed or name, plr)
    local isSystem = opts.system and true or false
    table.insert(f, { name = name, text = text, isMe = isMe and true or false,
        color = color, isSystem = isSystem })
    if #f > 60 then table.remove(f, 1) end
    if feed == activeTab and panelVisible() then
        renderBubble(name, text, isMe, color, isSystem)
    elseif not opts.silent then
        unread = unread + 1
        refreshFloat()
    end
end

local function storeSeparator(feed)
    local f = feeds[feed]
    if not f then return end
    table.insert(f, { sep = true })
    if #f > 60 then table.remove(f, 1) end
    if feed == activeTab and panelVisible() then renderSeparator() end
end

renderFeed = function(name)
    for _, fr in ipairs(frames) do pcall(function() fr:Destroy() end) end
    table.clear(frames)
    order = 0
    local feed = feeds[name]
    if not feed then return end
    for _, e in ipairs(feed) do
        if e.sep then
            renderSeparator()
        else
            renderBubble(e.name, e.text, e.isMe, e.color, e.isSystem)
        end
    end
end

local function paintTabs()
    for name, btn in pairs(tabBtns) do
        local sel = (name == activeTab)
        btn.BackgroundTransparency = sel and 0.08 or 1
        btn.TextColor3 = sel and Color3.fromRGB(255, 255, 255) or TEXT_DIM
        btn.Font = sel and Enum.Font.GothamBold or Enum.Font.Gotham
        local st = tabStrokes[name]
        if st then
            st.Color = sel and Color3.fromRGB(120, 190, 255) or Color3.fromRGB(255, 255, 255)
            st.Transparency = sel and 0.75 or 1
        end
        if name == "Private" then
            local dc = vezOnline and "0,200,0" or "255,60,60"
            btn.RichText = true
            btn.Text = string.format('<font color="rgb(%s)">●</font> Private', dc)
        end
    end
end

-- ================= Here channel discovery =================
local hereCh = nil
-- channels may live under TextChannels OR directly under TextChatService
local function allChannels()
    local out, seen = {}, {}
    local function scan(container)
        if not container then return end
        for _, ch in ipairs(container:GetChildren()) do
            if ch:IsA("TextChannel") and not seen[ch] then
                seen[ch] = true
                table.insert(out, ch)
            end
        end
    end
    pcall(function()
        scan(TCS:FindFirstChild("TextChannels"))
        scan(TCS)
    end)
    return out
end
local function channelRank(ch)
    local name = tostring(ch.Name)
    local lower = name:lower()
    if name == "RBXGeneral" then return 100 end
    if lower == "here" then return 90 end
    if lower == "general" or lower == "all" then return 80 end
    if lower:find("here", 1, true) then return 70 end
    if lower:find("general", 1, true) or lower:find("all", 1, true) then return 60 end
    if lower:find("global", 1, true) then return -100 end
    if lower:find("system", 1, true) then return -90 end
    if lower:find("team", 1, true) or lower:find("whisper", 1, true)
        or lower:find("private", 1, true) then
        return -50
    end
    return 10
end
local function pickHereChannel()
    local tcsFolder = TCS:FindFirstChild("TextChannels")
    local general = tcsFolder and tcsFolder:FindFirstChild("RBXGeneral")
    if general and general.Parent then return general end
    local best, bestRank = nil, -1000
    for _, ch in ipairs(allChannels()) do
        local rank = channelRank(ch)
        if rank > bestRank then best, bestRank = ch, rank end
    end
    if best and bestRank >= 0 then return best end
    return nil
end
local function findChannels()
    local ok, err = pcall(function()
        local pick = pickHereChannel()
        if pick ~= hereCh then
            if pick then
                LOG("chat", "here channel -> %s", tostring(pick.Name))
            elseif hereCh then
                LOG("chat", "here channel lost (was %s)", tostring(hereCh.Name))
            end
            hereCh = pick
        end
    end)
    if not ok then LOG("chat", "findChannels fail: %s", tostring(err)) end
end
local function getHere()
    findChannels()
    return (hereCh and hereCh.Parent) and hereCh or nil
end
task.spawn(function() -- warm in background (slow mobile replication)
    LOG("chat", "warming channels...")
    pcall(function() TCS:WaitForChild("TextChannels", 30) end)
    if not generationCurrent() then return end
    findChannels()
    local n = #allChannels()
    LOG("chat", "channels ready: %d (here=%s)", n, hereCh and hereCh.Name or "nil")
end)

local function guessSender()
    local other = nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then
            if other then return "Guest" end
            other = p
        end
    end
    if other then
        local dn = other.DisplayName
        if dn and dn ~= "" then return dn end
        return other.Name
    end
    return "Guest"
end

local function playerForName(name)
    if type(name) ~= "string" or name == "" then return nil end
    local exact, loose = {}, {}
    local low = name:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        local dn = p.DisplayName
        local un = p.Name
        if (dn and dn ~= "" and dn == name) or un == name then
            table.insert(exact, p)
        elseif (dn and dn ~= "" and dn:lower() == low) or un:lower() == low then
            table.insert(loose, p)
        end
    end
    if #exact == 1 then return exact[1] end
    if #exact == 0 and #loose == 1 then return loose[1] end
    return nil
end

    local cfg = TCS:FindFirstChildOfClass("BubbleChatConfiguration")

        font = Enum.Font.GothamMedium,

local function showPrivateBubble(plr, text)
    if not plr or type(text) ~= "string" then return end
    text = text:gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return end
    if #text > 200 then text = text:sub(1, 200) end
    local ch = plr.Character
    if not ch then return end
    local style = bubbleStyle()
    if style.enabled then
        local ok, err = pcall(function() TCS:DisplayBubble(ch, text) end)
        if ok then return end
        LOG("bubble", "DisplayBubble failed for %s: %s", plr.Name, tostring(err))
    elseif not bubbleDisabledLogged then
        bubbleDisabledLogged = true
        LOG("bubble", "bubble chat disabled; using custom bubbles")
    end
    local ok, err = pcall(showCustomBubble, ch, text, style)
    if not ok then LOG("bubble", "custom bubble failed for %s: %s", plr.Name, tostring(err)) end
end

-- The POST ack and the next poll can both contain the same message; suppress
-- the short-lived duplicate without hiding genuinely repeated chat later.
local vezRecent = {}
local lastVezPrune = 0
local function vezDuplicate(name, text)
    local now = os.clock()
    if now - lastVezPrune > 5 then
        lastVezPrune = now
        for k, t in pairs(vezRecent) do
            if now - t > 2 then vezRecent[k] = nil end
        end
    end
    local key = name .. "\0" .. text
    local t = vezRecent[key]
    if t and now - t < 2 then return true end
    vezRecent[key] = now
    return false
end

local function vezDisplay(raw, fromHistory)
    if type(raw) ~= "string" then return end
    raw = raw:gsub("^%s+", ""):gsub("%s+$", "")
    if raw == "" then return end
    if raw:find(HEARTBEAT, 1, true) then return end
    if raw:sub(1, 2) == "G|" then LOG("vez", "skip global message (tab removed)") return end
    LOG("vez", "recv: %s", raw:sub(1, 120))
    local name, text = raw:match("^(.-):%s*(.+)$")
    if not name then
        name, text = guessSender(), raw
    else
        name = name:gsub("%s+$", "")
        if name == "" then name = guessSender() end
    end
    if not fromHistory and vezDuplicate(name, text) then
        LOG("vez", "duplicate suppressed: %s", text:sub(1, 80))
        return
    end
    local isMe = (name == myName)
    local opts = fromHistory and { silent = true } or nil
    if isMe then
        storeMsg("Private", myName, text, true, lp.Name, lp, opts)
        if not fromHistory then showPrivateBubble(lp, text) end
    else
        local target = playerForName(name)
        if target == lp then target = nil end
        local seed = target and target.Name or name
        storeMsg("Private", name, text, false, seed, target, opts)
        if not fromHistory and target then showPrivateBubble(target, text) end
    end
end

local function handleBody(body)
    if not body or body == "" then return end
    local ok, data = pcall(HttpService.JSONDecode, HttpService, body)
    if not ok or type(data) ~= "table" then LOG("vez", "bad JSON body (%d bytes)", #body) return end
    local list = data.list or data.msgs
    if type(data.msg) == "string" then
        vezDisplay(data.msg)
    elseif type(list) == "table" then
        local initial = not privateHistoryDone
        if #list < shownUpTo then shownUpTo = 0 LOG("vez", "server list reset") end -- server cleared/restarted
        local start = shownUpTo + 1
        if initial and #list - start + 1 > 60 then
            start = #list - 59
            LOG("vez", "history trimmed to last %d msgs", #list - start + 1)
        end
        for i = start, #list do
            if type(list[i]) == "string" then vezDisplay(list[i], initial) end
        end
        shownUpTo = #list
        if initial then
            privateHistoryDone = true
            if #list > 0 then storeSeparator("Private") end
        end
    end
end

local function vezSend(text)
    local full = myName .. ": " .. text
    LOG("vez", "send: %s", text:sub(1, 120))
    task.spawn(function()
        local ok, res = doRequest(REMOTE, "POST", HttpService:JSONEncode({ m = full }))
        LOG("vez", "POST ok=%s", tostring(ok))
        if ok and res and res.Body then handleBody(res.Body) end
    end)
end

local pollStarted = false
local function startVezPoll()
    if pollStarted then return end
    pollStarted = true
    task.spawn(function()
        local fails = 0
        while generationCurrent() do
            pcall(findChannels) -- keep Here discovery fresh (late replication)
            local ok, res = doRequest(REMOTE_MSGS, "GET", nil)
            if not generationCurrent() then break end
            local online = ok and res and res.Success and res.Body and res.Body ~= "" and true or false
            if online ~= vezOnline then
                vezOnline = online
                LOG("vez", "online=%s", tostring(vezOnline))
            end
            if not online then
                fails = fails + 1
                if fails == 1 or fails % 10 == 0 then LOG("vez", "poll fail #%d ok=%s", fails, tostring(ok)) end
            else
                fails = 0
                pcall(handleBody, res.Body)
            end
            pcall(paintTabs)
            task.wait(1)
        end
    end)
end

local seenMsgIds = {}
local seenMsgOrder = {}
local seenMsgCount = 0
local SEEN_MSG_LIMIT = 500
local recentLegacy = {}
local lastLegacyPrune = 0

-- tracks our own sends so the server echo lands back in the tab it was sent from
local pendingOwn = {}
local function noteOwn(text, tab)
    local now = os.clock()
    for i = #pendingOwn, 1, -1 do
        if now - pendingOwn[i].t > 8 then table.remove(pendingOwn, i) end
    end
    table.insert(pendingOwn, { text = text, tab = tab, t = now })
end
local function consumeOwn(text)
    local now = os.clock()
    for i = #pendingOwn, 1, -1 do
        local e = pendingOwn[i]
        if now - e.t > 8 then
            table.remove(pendingOwn, i)
        elseif e.text == text then
            table.remove(pendingOwn, i)
            return e.tab
        end
    end
    return nil
end

-- cross-server senders aren't in this server: resolve names via userId (cached)
local nameCache = {}
local nameCacheOrder = {}
local nameCacheCount = 0
local NAME_CACHE_LIMIT = 200

-- System colors: the default RBXSystem styling wraps the text in a rich-text
-- <font color="..."> tag; games can also set ChatWindowMessageProperties or
-- legacy ExtraData.Color. Parse the real color and fall back to light gray.
local SYSTEM_COLOR = Color3.fromRGB(180, 180, 185)
local COLOR_NAMES = {
    white = { 255, 255, 255 }, black = { 0, 0, 0 }, red = { 255, 0, 0 },
    green = { 0, 255, 0 }, blue = { 0, 0, 255 }, yellow = { 255, 255, 0 },
    orange = { 255, 165, 0 }, gray = { 163, 162, 165 }, grey = { 163, 162, 165 },
    silver = { 163, 162, 165 },
}
local function parseCssColor(s)
    if type(s) ~= "string" then return nil end
    s = s:gsub("^%s+", ""):gsub("%s+$", "")
    local hex = s:match("^#(%x%x%x%x%x%x)$")
    if hex then
        return Color3.fromRGB(tonumber(hex:sub(1, 2), 16), tonumber(hex:sub(3, 4), 16), tonumber(hex:sub(5, 6), 16))
    end
    local r3, g3, b3 = s:match("^#(%x)(%x)(%x)$")
    if r3 then
        return Color3.fromRGB(tonumber(r3 .. r3, 16), tonumber(g3 .. g3, 16), tonumber(b3 .. b3, 16))
    end
    local r, g, b = s:match("^rgb%(%s*(%d+)%s*,%s*(%d+)%s*,%s*(%d+)%s*%)$")
    if r then return Color3.fromRGB(tonumber(r), tonumber(g), tonumber(b)) end
    local named = COLOR_NAMES[s:lower()]
    if named then return Color3.fromRGB(named[1], named[2], named[3]) end
    return nil
end
local function colorFromRich(text)
    if type(text) ~= "string" then return nil end
    local v = text:match('<font[^>]-color%s*=%s*"([^"]+)"')
        or text:match("<font[^>]-color%s*=%s*'([^']+)'")
    return parseCssColor(v)
end
local function systemColor(tm)
    local c = colorFromRich(tm.Text) or colorFromRich(tm.PrefixText)
    if c then return c end
    local ok, props = pcall(function() return tm.ChatWindowMessageProperties end)
    if ok and props then
        local ok2, pc = pcall(function() return props.TextColor3 end)
        if ok2 and typeof(pc) == "Color3" then return pc end
    end
    local meta = tm.Metadata
    if type(meta) == "string" and meta:lower():find("error", 1, true) then
        return Color3.fromRGB(255, 85, 85)
    end
    return nil
end
local function legacySystemColor(data)
    if type(data) ~= "table" then return nil end
    local extra = data.ExtraData
    if type(extra) == "table" then
        local ok, c = pcall(function() return extra.Color or extra.TextColor3 end)
        if ok and typeof(c) == "Color3" then return c end
    end
    local ok, c = pcall(function() return data.Color end)
    if ok and typeof(c) == "Color3" then return c end
    return nil
end

-- server system messages are shown in BOTH tabs, silently (no unread badge)
local function showSystem(text, source, color)
    if type(text) ~= "string" then return end
    local c = colorFromRich(text) or color
    if not c or typeof(c) ~= "Color3" then c = SYSTEM_COLOR end
    text = text:gsub("<[^>]*>", ""):gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return end
    LOG("chat", "system[%s] color=%d,%d,%d: %s", tostring(source),
        math.floor(c.R * 255), math.floor(c.G * 255), math.floor(c.B * 255), text:sub(1, 100))
    local opts = { system = true, color = c, silent = true }
    storeMsg("Normal", "System", text, false, nil, nil, opts)
    storeMsg("Private", "System", text, false, nil, nil, opts)
end

-- plr == nil + userId == nil means a system message.
-- The legacy [GLOBAL] tag is stripped; messages land in Normal.
local function routeRoblox(plr, text, metadata, channel, userId, sysColor)
    if type(text) ~= "string" then return end
    text = text:gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return end
    if text:sub(1, #GLOBAL_TAG) == GLOBAL_TAG then
        text = text:sub(#GLOBAL_TAG + 1):gsub("^%s+", "")
        if text == "" then return end
    end
    if not plr then
        if userId and userId > 0 then
            task.spawn(function()
                local nm = nameCache[userId]
                if not nm then
                    local ok, un = pcall(function() return Players:GetNameFromUserIdAsync(userId) end)
                    nm = (ok and un and un ~= "") and un or ("User" .. tostring(userId))
                    nameCache[userId] = nm
                    nameCacheCount = nameCacheCount + 1
                    table.insert(nameCacheOrder, userId)
                    if nameCacheCount > NAME_CACHE_LIMIT then
                        local old = table.remove(nameCacheOrder, 1)
                        if old then
                            nameCache[old] = nil
                            nameCacheCount = nameCacheCount - 1
                        end
                    end
                end
                if not generationCurrent() then return end
                storeMsg("Normal", nm, text, false, nm)
            end)
        else
            showSystem(text, "service", sysColor)
        end
        return
    end
    local isMe = (plr == lp)
    local dn = plr.DisplayName
    local nm = ((dn and dn ~= "") and dn) or plr.Name
    if isMe then
        -- echo of our own send: back to its tab (or Normal if typed in default UI)
        local tab = consumeOwn(text)
        storeMsg(tab or "Normal", nm, text, true, lp.Name, lp)
        return
    end
    storeMsg("Normal", nm, text, false, plr.Name, plr)
end

-- modern TextChatService path (official pattern per Roblox docs)
local function seenIds(tm)
    local id = tm.MessageId
    if type(id) ~= "string" or id == "" then return false end
    if seenMsgIds[id] then return true end
    seenMsgIds[id] = true
    seenMsgCount = seenMsgCount + 1
    table.insert(seenMsgOrder, id)
    if seenMsgCount > SEEN_MSG_LIMIT then
        local old = table.remove(seenMsgOrder, 1)
        if old then
            seenMsgIds[old] = nil
            seenMsgCount = seenMsgCount - 1
        end
    end
    return false
end
do
    local ok, err = pcall(function()
        TCS.MessageReceived:Connect(function(tm)
            if not generationCurrent() then return end
            local iok, ierr = pcall(function()
                if not tm then return end
                if tm.Status == Enum.TextChatMessageStatus.Sending then return end
                if seenIds(tm) then return end
                local txt = tm.Text
                if type(txt) ~= "string" or txt == "" then return end
                local plr, uid = nil, nil
                if tm.TextSource then
                    uid = tm.TextSource.UserId
                    if uid and uid > 0 then
                        pcall(function() plr = Players:GetPlayerByUserId(uid) end)
                    else
                        uid = nil
                    end
                end
                routeRoblox(plr, txt, tm.Metadata, tm.TextChannel, uid,
                    (not plr and not uid) and systemColor(tm) or nil)
            end)
            if not iok then LOG("chat", "MessageReceived handler fail: %s", tostring(ierr)) end
        end)
    end)
    LOG("chat", "MessageReceived hook ok=%s err=%s", tostring(ok), tostring(err))
end
-- legacy chat fallback (NON-BLOCKING: old code yielded 5s+5s on main thread -> looked like load fail)
task.spawn(function()
    local ok, err = pcall(function()
        LOG("chat", "checking legacy DefaultChatSystemChatEvents...")
        local evs = ReplicatedStorage:WaitForChild("DefaultChatSystemChatEvents", 5)
        if not generationCurrent() then return end
        if not evs then LOG("chat", "no legacy chat (modern game)") return end
        local done = evs:FindFirstChild("OnMessageDoneFiltering")
        if done then
            done.OnClientEvent:Connect(function(data)
                if not generationCurrent() then return end
                if type(data) ~= "table" then return end
                local txt = data.Message
                if type(txt) ~= "string" or txt == "" then return end
                local key = tostring(data.FromSpeaker) .. "\0" .. txt
                local now = os.clock()
                if now - lastLegacyPrune > 5 then
                    lastLegacyPrune = now
                    for k, t in pairs(recentLegacy) do
                        if now - t > 3 then recentLegacy[k] = nil end
                    end
                end
                if recentLegacy[key] and (now - recentLegacy[key]) < 3 then return end
                recentLegacy[key] = now
                local plr = nil
                local sid = tonumber(data.SpeakerUserId or "")
                if sid and sid > 0 then
                    plr = Players:GetPlayerByUserId(sid)
                elseif data.FromSpeaker and data.FromSpeaker ~= "" and data.FromSpeaker ~= "System" then
                    plr = Players:FindFirstChild(data.FromSpeaker)
                end
                routeRoblox(plr, txt, nil, nil, sid, legacySystemColor(data))
            end)
            LOG("chat", "legacy player hook attached")
        end
        -- system/announcement messages arrive on a separate legacy event
        local sys = evs:FindFirstChild("OnNewSystemMessage") or evs:FindFirstChild("SystemMessage")
        if sys then
            sys.OnClientEvent:Connect(function(data)
                if not generationCurrent() then return end
                local txt = data
                if type(data) == "table" then txt = data.Message or data.Text end
                if type(txt) == "string" then showSystem(txt, "legacy", legacySystemColor(data)) end
            end)
            LOG("chat", "legacy system hook attached (%s)", sys.Name)
        end
        if not done and not sys then LOG("chat", "no legacy chat events found") end
    end)
    if not ok then LOG("chat", "legacy hook fail: %s", tostring(err)) end
end)

-- Normal goes to the Here channel; legacy fallback if TextChatService is absent.
local function robloxSend(text, tab)
    LOG("chat", "robloxSend tab=%s len=%d", tab, #text)
    local h = getHere()
    if h then
        local ok = pcall(function() h:SendAsync(text) end)
        LOG("chat", "send here ok=%s", tostring(ok))
        if ok then return true, text end
    end
    local out = text
    local sent = false
    pcall(function()
        local evs = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
        local req = evs and evs:FindFirstChild("SayMessageRequest")
        if req then req:FireServer(out, "All") sent = true end
    end)
    LOG("chat", "legacy send sent=%s", tostring(sent))
    if sent then return true, out end
    return false, nil
end

local function setLocked(v)
    locked = v and true or false
    LOG("ui", "setLocked=%s", tostring(locked))
    local okSave, info = pcall(savePillPos)
    local saveInfo = " Saved: " .. (okSave and tostring(info) or "fail") .. "."
    storeMsg(activeTab, "System",
        (locked and "Locked: panel + chat icon frozen (/unlock to move)"
            or "Unlocked: drag header / pill to move (/lock to freeze)") .. saveInfo, false)
end

-- /l: send the icon's + panel's current (locked) position to the webhook
local function logIconPos()
    local ax, ay = floatAbs()
    local px, py = panelAbs()
    local vs = viewport()
    local p = floatBtn.Position
    local pp = main.Position
    local tab = activeTab
    local payload = {
        command = "/l",
        locked = locked,
        icon = {
            absoluteX = math.floor(ax * 10 + 0.5) / 10,
            absoluteY = math.floor(ay * 10 + 0.5) / 10,
            scaleX = p.X.Scale,
            scaleY = p.Y.Scale,
            offsetX = p.X.Offset,
            offsetY = p.Y.Offset,
        },
        panel = {
            absoluteX = math.floor(px * 10 + 0.5) / 10,
            absoluteY = math.floor(py * 10 + 0.5) / 10,
            scaleX = pp.X.Scale,
            scaleY = pp.Y.Scale,
            offsetX = pp.X.Offset,
            offsetY = pp.Y.Offset,
        },
        viewport = { width = vs.X, height = vs.Y },
        placeId = tostring(game.PlaceId),
        jobId = tostring(game.JobId),
        player = myName,
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
    }
    task.spawn(function()
        local ok, body = pcall(HttpService.JSONEncode, HttpService, payload)
        if not ok then
            LOG("webhook", "/l JSONEncode fail: %s", tostring(body))
            storeMsg(tab, "System", "/l failed (json encode)", false)
            return
        end
        local callOK, reqOK, res = pcall(doRequest, WEBHOOK_URL, "POST", body)
        local sent = callOK and reqOK
        local code = 0
        if sent and type(res) == "table" then
            code = res.StatusCode or (res.Success and 200) or 0
        end
        LOG("webhook", "/l send ok=%s code=%s icon=%d,%d locked=%s",
            tostring(sent), tostring(code), math.floor(ax), math.floor(ay), tostring(locked))
        storeMsg(tab, "System", string.format("/l logged icon %d,%d (locked=%s, http=%s)",
            math.floor(ax), math.floor(ay), tostring(locked), sent and tostring(code) or "fail"), false)
    end)
end

local function sendText(text)
    if not text then return end
    -- TextBox:FocusLost can deliver a trailing newline; never forward it as a
    -- second empty chat line (or as part of a VEZ payload).
    local clean = text:gsub("[\r\n]+", " "):gsub("^%s+", ""):gsub("%s+$", "")
    if clean == "" then return end
    local cmd = clean:lower()
    if cmd == "/lock" then setLocked(true) return end
    if cmd == "/unlock" then setLocked(false) return end
    if cmd == "/l" then logIconPos() return end
    if activeTab == "Private" then
        vezSend(clean)
    else
        local tab = activeTab
        task.spawn(function()
            -- record the EXACT outgoing text so the echo matches back to this tab
            local ok, out = robloxSend(clean, tab)
            if ok and out then
                noteOwn(out, tab)
            else
                LOG("chat", "send FAILED tab=%s", tab)
                storeMsg(tab, "System", "Couldn't send (Roblox blocked it)", false)
            end
        end)
    end
end

sendBtn.MouseButton1Click:Connect(function()
    sendText(box.Text)
    box.Text = ""
end)
box.FocusLost:Connect(function(enter)
    if enter then
        sendText(box.Text)
        box.Text = ""
    end
end)
